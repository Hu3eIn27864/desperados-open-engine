cmake_minimum_required(VERSION 3.25)
project(DesperadosOpenEngine VERSION 0.1.0 LANGUAGES CXX)
option(DOE_BUILD_GAME "Build the SDL3 application" OFF)
option(DOE_BUILD_TESTS "Build core unit tests" ON)
add_library(doe_core src/core/binary_reader.cpp)
add_library(doe::core ALIAS doe_core)
target_include_directories(doe_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>)
target_compile_features(doe_core PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(doe_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(doe_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion)
endif()
if(DOE_BUILD_GAME)
  find_package(SDL3 CONFIG REQUIRED)
  add_executable(desperados-open-engine apps/game/main.cpp)
  target_link_libraries(desperados-open-engine PRIVATE doe::core SDL3::SDL3)
  target_compile_features(desperados-open-engine PRIVATE cxx_std_20)
endif()
if(DOE_BUILD_TESTS)
  enable_testing()
  add_executable(doe_core_tests tests/unit/binary_reader_tests.cpp)
  target_link_libraries(doe_core_tests PRIVATE doe::core)
  add_test(NAME doe_core_tests COMMAND doe_core_tests)
endif()
